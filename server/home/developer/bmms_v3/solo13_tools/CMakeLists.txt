cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(bmms_solo5 LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT DEFINED NPU_ARCH)
    set(NPU_ARCH dav-2201)
endif()
foreach(program tiling_inspector)
    add_executable(${program} ${program}.asc)
    target_include_directories(${program} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(${program} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
    target_link_libraries(${program} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endforeach()
