cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(bmms_precision LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT DEFINED NPU_ARCH)
    set(NPU_ARCH dav-2201)
endif()
add_executable(bench_cases bench_cases.asc)
target_include_directories(bench_cases PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_options(bench_cases PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
target_link_libraries(bench_cases PRIVATE tiling_api register platform unified_dlog dl m graph_base)
